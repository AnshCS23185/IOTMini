# INA219 CONFIGURATION
# ============================================================

INA1 = 0x40
INA2 = 0x41

REG_CONFIG = 0x00
REG_SHUNT = 0x01
REG_BUS = 0x02
REG_CURRENT = 0x04
REG_CALIBRATION = 0x05


# ============================================================
# INA219 WRITE REGISTER
# ============================================================

def write_reg(
    address,
    register,
    value
):

    data = bytearray(2)

    data[0] = (
        value >> 8
    ) & 0xFF

    data[1] = (
        value
        & 0xFF
    )

    i2c.writeto_mem(
        address,
        register,
        data
    )


# ============================================================
# INA219 READ REGISTER
# ============================================================

def read_reg(
    address,
    register
):

    data = i2c.readfrom_mem(
        address,
        register,
        2
    )

    return (
        (data[0] << 8)
        | data[1]
    )


# ============================================================
# INA219 SIGNED REGISTER
# ============================================================

def read_signed(
    address,
    register
):

    value = read_reg(
        address,
        register
    )

    if value & 0x8000:

        value -= 65536

    return value


# ============================================================
# INA219 SETUP
# ============================================================

def setup_ina219(
    address
):

    # 32 V bus range
    # Continuous bus + shunt measurement

    write_reg(
        address,
        REG_CONFIG,
        0x399F
    )


    # Calibration for approximately
    # 0.1 ohm shunt

    write_reg(
        address,
        REG_CALIBRATION,
        4096
    )


# ============================================================
# INA219 READ
# ============================================================

def read_ina219(
    address
):

    # --------------------------------------------------------
    # BUS VOLTAGE
    # --------------------------------------------------------

    raw_bus = read_reg(
        address,
        REG_BUS
    )

    raw_bus >>= 3

    voltage = (
        raw_bus
        * 0.004
    )


    # --------------------------------------------------------
    # SHUNT VOLTAGE
    # --------------------------------------------------------

    raw_shunt = read_signed(
        address,
        REG_SHUNT
    )

    shunt_voltage = (
        raw_shunt
        * 0.00001
    )


    # --------------------------------------------------------
    # CURRENT
    # --------------------------------------------------------

    raw_current = read_signed(
        address,
        REG_CURRENT
    )

    current = (
        raw_current
        * 0.0001
    )


    # --------------------------------------------------------
    # POWER
    # --------------------------------------------------------

    power = (
        voltage
        * current
    )


    return (
        voltage,
        current,
        power
    )

# ============================================================
# NTP TIME SYNCHRONIZATION
# ============================================================

def sync_time():

    print()
    print(
        "Synchronizing time with NTP..."
    )


    for attempt in range(3):

        try:

            print(
                "NTP attempt",
                attempt + 1,
                "of 3"
            )


            ntptime.host = (
                "pool.ntp.org"
            )


            # Synchronize Pico RTC
            # with UTC.

            ntptime.settime()


            # Read synchronized time.

            t = time.gmtime()


            print()
            print(
                "TIME SYNCHRONIZATION SUCCESS"
            )


            print(
                "UTC:",
                "{:04d}-{:02d}-{:02d} "
                "{:02d}:{:02d}:{:02d}".format(
                    t[0],
                    t[1],
                    t[2],
                    t[3],
                    t[4],
                    t[5]
                )
            )


            return True


        except Exception as e:

            print(
                "NTP ERROR:",
                e
            )


            if attempt < 2:

                time.sleep(2)


    print()

    print(
        "TIME SYNCHRONIZATION FAILED"
    )

    return False

def device_headers():

# ============================================================
# TIMESTAMP
# ============================================================
#
# Returns UTC timestamp.
#
# Example:
#
# 2026-09-07T10:30:00Z
#
# Z = UTC
# ============================================================

def get_timestamp():

    try:

        t = time.gmtime()


        return (
            "{:04d}-{:02d}-{:02d}T"
            "{:02d}:{:02d}:{:02d}Z"
        ).format(
            t[0],
            t[1],
            t[2],
            t[3],
            t[4],
            t[5]
        )


    except Exception:

        return (
            "1970-01-01T00:00:00Z"
        )

# ============================================================
# POLL PENDING COMMANDS
# ============================================================

def poll_commands():

    response = None


    try:

        url = (
            API_BASE_URL
            + "/iot/devices/"
            + DEVICE_UID
            + "/commands?status=PENDING"
        )


        response = urequests.get(
            url,
            headers=device_headers()
        )


        print()

        print(
            "COMMAND POLL HTTP:",
            response.status_code
        )


        if response.status_code != 200:

            print(
                "COMMAND POLL FAILED:",
                response.text
            )


            response.close()

            response = None

            return


        commands = response.json()


        response.close()

        response = None


        if not commands:

            print(
                "No pending commands."
            )

            return


        print(
            "Pending commands:",
            len(commands)
        )


        for command in commands:

            process_command(
                command
            )


    except Exception as e:

        print()

        print(
            "COMMAND POLLING ERROR:",
            e
        )


        if response is not None:

            try:

                response.close()

            except:

                pass


# ============================================================
# PROCESS COMMAND
# ============================================================

def process_command(
    command
):

    try:

        command_id = command.get(
            "id"
        )

        panel_id = command.get(
            "panel_id"
        )

        command_type = command.get(
            "command"
        )


        print()
        print(
            "------------------------------"
        )

        print(
            "RECEIVED COMMAND"
        )

        print(
            "------------------------------"
        )


        print(
            "Command ID :",
            command_id
        )


        print(
            "Panel ID   :",
            panel_id
        )


        print(
            "Command    :",
            command_type
        )


        # ====================================================
        # HARDWARE SAFETY CHECK
        # ====================================================
        #
        # ONLY CH2 CAN CONTROL RELAY 2.
        #
        # CH1 IS NEVER ALLOWED.
        # ====================================================

        if panel_id != CHANNEL2:

            print()

            print(
                "SAFETY: Ignoring command because it is not CH2."
            )


            acknowledge_command(
                command_id,
                "FAILED",
                "Only CH2 / Relay 2 is remotely controllable."
            )


            return


        # ====================================================
        # RELAY 2 ON
        # ====================================================

        if command_type == "RELAY_ON":

            relay2.value(0)


            print()

            print(
                "RELAY 2 : ON"
            )


            acknowledge_command(
                command_id,
                "ACKNOWLEDGED"
            )


            return


        # ====================================================
        # RELAY 2 OFF
        # ====================================================

        elif command_type == "RELAY_OFF":

            relay2.value(1)


            print()

            print(
                "RELAY 2 : OFF"
            )


            acknowledge_command(
                command_id,
                "ACKNOWLEDGED"
            )


            return


        # ====================================================
        # UNKNOWN COMMAND
        # ====================================================

        else:

            print()

            print(
                "UNKNOWN COMMAND:",
                command_type
            )


            acknowledge_command(
                command_id,
                "FAILED",
                "Unsupported relay command."
            )


    except Exception as e:

        print()

        print(
            "COMMAND PROCESSING ERROR:",
            e
        )


        try:

            if command.get("id") is not None:

                acknowledge_command(
                    command.get("id"),
                    "FAILED",
                    str(e)
                )

        except:

            pass


# ============================================================
# ACKNOWLEDGE COMMAND
# ============================================================

def acknowledge_command(
    command_id,
    status,
    error_message=None
):

    response = None


    try:

        url = (
            API_BASE_URL
            + "/iot/devices/"
            + DEVICE_UID
            + "/commands/"
            + str(command_id)
            + "/ack"
        )


        payload = {
            "status": status
        }


        if (
            status == "FAILED"
            and error_message is not None
        ):

            payload[
                "error_message"
            ] = error_message


        print()

        print(
            "Acknowledging command:",
            command_id
        )


        response = urequests.post(
            url,
            headers=device_headers(),
            json=payload
        )


        print(
            "ACK HTTP:",
            response.status_code
        )


        if (
            response.status_code >= 200
            and response.status_code < 300
        ):

            print(
                "Command",
                command_id,
                "->",
                status
            )

        else:

            print(
                "ACK FAILED:",
                response.text
            )


        response.close()

        response = None


    except Exception as e:

        print()

        print(
            "ACK ERROR:",
            e
        )


        if response is not None:

            try:

                response.close()

            except:

                pass

            ldr_value1
        )


        # ----------------------------------------------------
        # PANEL 2
        # ----------------------------------------------------

        send_panel_reading(
            CHANNEL2,
            voltage2,
            current2,
            power2,
            temperature2,
            humidity2,
            ldr_value2
        )


    else:

        print()

        print(
            "Skipping telemetry upload - Wi-Fi unavailable."
        )


    # ========================================================
    # HEARTBEAT
    # ========================================================

    current_time = (
        time.ticks_ms()
    )


    if time.ticks_diff(
        current_time,
        last_heartbeat
    ) >= (
        HEARTBEAT_INTERVAL
        * 1000
    ):

        last_heartbeat = (
            current_time
        )


        if wifi_connected:

            send_heartbeat()


    # ========================================================
    # PERIODIC TIME SYNCHRONIZATION
    # ========================================================
    #
    # Re-synchronize every 6 hours.
    # ========================================================

    current_time = (
        time.ticks_ms()
    )


    if time.ticks_diff(
        current_time,
        last_time_sync
    ) >= (
        TIME_SYNC_INTERVAL
        * 1000
    ):

        last_time_sync = (
            current_time
        )


        if wifi_connected:

            if sync_time():

                time_synchronized = True

            else:

                time_synchronized = False


    # ========================================================
    # FINAL RELAY SAFETY CHECK

    )

        "Relay 2 :",
        "ON"
        if relay2.value() == 0
        else "OFF"
    )

    # ========================================================

    print()

    print(
        "Current timestamp:",
        get_timestamp()
    )


    print()

        "Next telemetry cycle in 10 seconds..."
    )

    # WAIT
    # ========================================================

    time.sleep(
        TELEMETRY_INTERVAL
